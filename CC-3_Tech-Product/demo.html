A job is a function

Wrap side effects in step(). Spool stores the result, so replays skip what already ran.

export const onboard = spool.job("user.onboard",
  async ({ user, step }) => {
    await step("welcome", () =>
      email.send(user, "welcome"))

    await step.sleep("3d")  // survives deploys

    const used = await step("check",
      () => usage.of(user))

    if (!used) await step("nudge",
      () => email.send(user, "tips"))
})
# replay a failed run from step 3
$ spool replay run_8f2a1c --from check

# tail a queue in your terminal
$ spool tail emails.send --failed
